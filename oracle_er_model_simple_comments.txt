-- student table
CREATE TABLE Student154 (
    roll_no NUMBER,
    sname VARCHAR2(100) NOT NULL,
    email_id VARCHAR2(100),
    mobile_no NUMBER(10,0),
    date_of_birth DATE,

    CONSTRAINT pk_student PRIMARY KEY (roll_no),
    CONSTRAINT uq_student_email UNIQUE (email_id),
    CONSTRAINT uq_student_mobile UNIQUE (mobile_no)
);

-- course table
CREATE TABLE Course154 (
    cid NUMBER,
    cname VARCHAR2(100) NOT NULL,
    discipline VARCHAR2(100),

    CONSTRAINT pk_course PRIMARY KEY (cid)
);

-- class section table
CREATE TABLE Class_Section154 (
    section_id NUMBER,
    batch NUMBER,

    CONSTRAINT pk_class_section PRIMARY KEY (section_id)
);

-- subject paper table
CREATE TABLE SubjectPaper154 (
    paper_code NUMBER,
    pname VARCHAR2(100) NOT NULL,
    credit NUMBER,
    cid NUMBER,
    semester NUMBER,

    CONSTRAINT pk_subject_paper PRIMARY KEY (paper_code)
);

-- faculty table
CREATE TABLE Faculty154 (
    emp_id NUMBER,
    name VARCHAR2(100) NOT NULL,
    mobile_no VARCHAR2(15),
    salary NUMBER(10,2),
    did NUMBER,
    supervisor_emp_id NUMBER,

    CONSTRAINT pk_faculty PRIMARY KEY (emp_id),
    CONSTRAINT uq_faculty_mobile UNIQUE (mobile_no)
);

-- department table
CREATE TABLE Department154 (
    did NUMBER,
    dname VARCHAR2(100) NOT NULL,
    office_location VARCHAR2(100),
    year_started NUMBER(4),
    head_emp_id NUMBER,

    CONSTRAINT pk_department PRIMARY KEY (did)
);

-- exam table
CREATE TABLE Exam154 (
    roll_no NUMBER,
    paper_code NUMBER,
    session VARCHAR2(20),
    marks NUMBER(5,2),

    CONSTRAINT pk_exam
        PRIMARY KEY (roll_no, paper_code, session),

    CONSTRAINT fk_exam_student
        FOREIGN KEY (roll_no)
        REFERENCES Student154(roll_no),

    CONSTRAINT fk_exam_subject_paper
        FOREIGN KEY (paper_code)
        REFERENCES SubjectPaper154(paper_code)
);

-- teaches table
CREATE TABLE Teaches154 (
    emp_id NUMBER,
    paper_code NUMBER,
    session VARCHAR2(20),

    CONSTRAINT pk_teaches
        PRIMARY KEY (emp_id, paper_code, session),

    CONSTRAINT fk_teaches_faculty
        FOREIGN KEY (emp_id)
        REFERENCES Faculty154(emp_id),

    CONSTRAINT fk_teaches_subject_paper
        FOREIGN KEY (paper_code)
        REFERENCES SubjectPaper154(paper_code)
);

-- check foreign keys
SELECT
    uc.table_name AS child_table,
    ucc.column_name AS child_column,
    uc.constraint_name,
    r_uc.table_name AS parent_table,
    r_ucc.column_name AS parent_column
FROM user_constraints uc
JOIN user_cons_columns ucc
    ON uc.constraint_name = ucc.constraint_name
    AND uc.owner = ucc.owner
JOIN user_constraints r_uc
    ON uc.r_constraint_name = r_uc.constraint_name
    AND uc.owner = r_uc.owner
JOIN user_cons_columns r_ucc
    ON r_uc.constraint_name = r_ucc.constraint_name
    AND r_uc.owner = r_ucc.owner
WHERE uc.constraint_type = 'R'
ORDER BY uc.table_name;

-- check primary keys

    table_name,
    constraint_name

WHERE constraint_type = 'P'
ORDER BY table_name;

-- check foreign keys
SELECT
    table_name,
    constraint_name
FROM user_constraints
WHERE constraint_type = 'R'
ORDER BY table_name;

-- check last change

    lt.table_name,
    rt.column_name,
    lt.constraint_name,
    lt.constraint_type,
    lt.search_condition,
    lt.last_change

     user_cons_columns rt
WHERE lt.constraint_name = rt.constraint_name
AND lt.last_change >= TO_DATE('14-AUG-2026', 'DD-MON-YYYY')
ORDER BY lt.table_name;
